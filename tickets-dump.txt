58: /** ISO string → "YYYY-MM-DDTHH:mm" for <input type="datetime-local">. */
59: function toLocalInputValue(iso: string): string {
60:   const d = new Date(iso);
61:   if (Number.isNaN(d.getTime())) return '';
62:   const pad = (n: number) => String(n).padStart(2, '0');
63:   return `${d.getFullYear()}-${pad(d.getMonth() + 1)}-${pad(d.getDate())}T${pad(d.getHours())}:${pad(d.getMinutes())}`;
64: }
